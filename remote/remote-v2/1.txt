```python
    def _apply_smoothing(self, words_with_speaker, duration_threshold=0.8):
        """
        平滑修正（三明治法则）：修复 A -> B -> A 的短时跳变错误
        
        Args:
            words_with_speaker: 带说话人标签的词列表
            duration_threshold: 时长阈值（秒），推荐 0.8s。
                                既能覆盖 0.26s 的错误，又能保留 >0.8s 的真实短句。
        """
        if len(words_with_speaker) < 3:
            return words_with_speaker

        smoothed_count = 0
        
        # 定义"强语义保护词"（可选）：这些词即使很短，也不要轻易合并
        # protected_words = {'不', '没', '停', '错'} 
        
        for i in range(1, len(words_with_speaker) - 1):
            prev_word = words_with_speaker[i-1]
            curr_word = words_with_speaker[i]
            next_word = words_with_speaker[i+1]

            # 1. 基础字段检查
            if 'speaker' not in prev_word or 'speaker' not in next_word:
                continue

            speaker_prev = prev_word['speaker']
            speaker_curr = curr_word['speaker']
            speaker_next = next_word['speaker']

            # 2. 检查夹心结构: A -> B -> A
            if speaker_prev == speaker_next and speaker_curr != speaker_prev:
                
                duration = curr_word['end'] - curr_word['start']
                text = curr_word['text']
                
                # 3. 核心判断：时长 < 阈值
                # (如果未来需要保护特定词，可以在这里加: if duration < threshold and text not in protected_words)
                if duration < duration_threshold:
                    old_speaker = speaker_curr
                    
                    # 执行修正：将中间的 B 同化为 A
                    words_with_speaker[i]['speaker'] = speaker_prev
                    words_with_speaker[i]['corrected'] = True
                    words_with_speaker[i]['correction_reason'] = 'smoothing_sandwich'
                    
                    self.logger.info(
                        f"🔧 平滑修正: '{text}' "
                        f"[{old_speaker} -> {speaker_prev}] "
                        f"(时长={duration:.2f}s < {duration_threshold}s, 原因=A-B-A结构)"
                    )
                    smoothed_count += 1

        if smoothed_count > 0:
            self.logger.info(f"✅ 平滑修正完成: 修正了 {smoothed_count} 个碎片")
        
        return words_with_speaker
```
