### 第二步：核心算法 (`src/utils/audio_utils.py`)

**核心变更**：
1.  **VAD 信任逻辑**：`if user != target -> Customer`。
2.  **网格平滑**：填补 0.5s 内的空隙，剔除 < 0.3s 的孤立片段（过滤掉偶尔的咳嗽/撞击声）。
3.  **分数映射**：确保输出 0-100。

def map_score_to_percentage(z_score, high_threshold=3.5, low_threshold=1.8):
    """
    将 Z-Score 映射到 0-100 的数值 (人性化显示)
    """
    z_score = float(z_score)
    
    if z_score < low_threshold:
        # 低分区间: 映射到 0 ~ 60 分
        # 假设最低分是 -5.0 (给足余量)
        min_z = -5.0
        if z_score < min_z: z_score = min_z
        
        percent = 60 * (z_score - min_z) / (low_threshold - min_z)
        return max(0.0, min(60.0, percent))
    
    elif z_score > high_threshold:
        # 高分区间: 映射到 90 ~ 99.9 分
        diff = z_score - high_threshold
        percent = 90 + 9.9 * (1 - 1 / (1 + 0.5 * diff))
        return min(99.9, percent)
        
    else:
        # 中间区间: 映射到 60 ~ 90 分
        ratio = (z_score - low_threshold) / (high_threshold - low_threshold)
        percent = 60 + 30 * ratio
        return percent

def calculate_diarization_intervals(raw_segments, target_uid, total_duration, stride=0.1, max_silence_gap=15.0):
    """
    V10.0 全局说话人区分 (VAD 信任 + 网格平滑)
    """
    # 1. 初始化网格 (Grid)
    # 0: Silence, 1: Target, 2: Customer
    num_steps = int(np.ceil(total_duration / stride)) + 1
    grid = np.zeros(num_steps, dtype=int)
    grid_scores = np.zeros(num_steps, dtype=float)

    # 2. 填充网格 (Voting)
    if raw_segments:
        for seg in raw_segments:
            center_time = seg['start'] + (seg['end'] - seg['start']) / 2
            idx = int(center_time / stride)
            
            if 0 <= idx < num_steps:
                current_val = grid[idx]
                
                # 逻辑：Target 优先，否则只要是 raw_segments 里的(VAD过的)就是 Customer
                if seg['user'] == target_uid:
                    grid[idx] = 1
                    grid_scores[idx] = seg['score']
                else:
                    # 如果当前还没被标记为 Target，就标记为 Customer
                    if current_val != 1:
                        grid[idx] = 2
                        grid_scores[idx] = seg['score']

    # 3. 网格平滑 (Smoothing) - 消除碎片与噪音
    
    # A. 填补 Target 空隙 (防止员工说话断断续续)
    # 允许填补 0.5s (5步) 的空隙
    for i in range(1, num_steps - 1):
        if grid[i] != 1:
            left = max(0, i - 5)
            right = min(num_steps, i + 6)
            if 1 in grid[left:i] and 1 in grid[i+1:right]:
                grid[i] = 1
                # 分数补全
                neighbors = [s for s in grid_scores[left:right] if s != 0]
                if neighbors: grid_scores[i] = sum(neighbors)/len(neighbors)

    # B. 填补 Customer 空隙
    for i in range(1, num_steps - 1):
        if grid[i] == 0: # 只有静音才填补
            left = max(0, i - 5)
            right = min(num_steps, i + 6)
            if 2 in grid[left:i] and 2 in grid[i+1:right]:
                grid[i] = 2
                neighbors = [s for s in grid_scores[left:right] if s != 0]
                if neighbors: grid_scores[i] = sum(neighbors)/len(neighbors)

    # C. 剔除极短的孤立片段 (噪音过滤)
    # 如果一段声音 < 0.3s (3步)，视为 VAD 误报(如撞击声)，抹除为静音
    i = 0
    while i < num_steps:
        val = grid[i]
        if val != 0:
            j = i
            while j < num_steps and grid[j] == val:
                j += 1
            length = j - i
            if length < 3: # < 0.3s
                grid[i:j] = 0
            i = j
        else:
            i += 1

    # 4. 网格转片段 (Grid -> Intervals)
    final_results = []
    if num_steps > 0:
        current_type = grid[0]
        start_idx = 0
        current_scores = [grid_scores[0]] if grid_scores[0] != 0 else []

        for i in range(1, num_steps):
            val = grid[i]
            if val != current_type:
                # 结算上一段
                _append_segment(final_results, current_type, start_idx, i, current_scores, target_uid, stride)
                # 开启新一段
                current_type = val
                start_idx = i
                current_scores = []
            
            if val != 0:
                current_scores.append(grid_scores[i])
        
        # 结算最后一段
        _append_segment(final_results, current_type, start_idx, num_steps, current_scores, target_uid, stride)

    # 5. 合并同类项 (Merge)
    merged_results = []
    if final_results:
        curr = final_results[0]
        for i in range(1, len(final_results)):
            next_seg = final_results[i]
            
            # 类型相同则尝试合并
            if curr['type'] == next_seg['type']:
                gap = next_seg['start'] - curr['end']
                
                # Target: 允许长静音 (max_silence_gap)
                if curr['type'] == 'target' and gap <= max_silence_gap:
                    curr['end'] = next_seg['end']
                    curr['scores'].extend(next_seg['scores'])
                    curr['window_count'] += next_seg['window_count']
                    continue
                
                # Customer: 允许短静音 (2.0s)
                if curr['type'] == 'other' and gap <= 2.0:
                    curr['end'] = next_seg['end']
                    curr['scores'].extend(next_seg['scores'])
                    curr['window_count'] += next_seg['window_count']
                    continue
                
                # Silence: 直接合并
                if curr['type'] == 'silence':
                    curr['end'] = next_seg['end']
                    continue

            merged_results.append(curr)
            curr = next_seg
        merged_results.append(curr)

    # 6. 格式化输出 (应用百分比映射)
    output = []
    for seg in merged_results:
        if seg['scores']:
            avg_z = sum(seg['scores']) / len(seg['scores'])
        else:
            avg_z = 0.0
            
        # 映射百分比 (0-100)
        # 无论是 Target 还是 Customer，都计算一个分数展示
        percent = map_score_to_percentage(avg_z, high_threshold=3.5, low_threshold=1.8)
            
        output.append({
            "start": round(seg['start'], 2),
            "end": round(seg['end'], 2),
            "user": seg['user'],
            "type": seg['type'],
            "score": round(percent, 1), # 0-100
            "raw_z_score": round(avg_z, 2),
            "window_count": seg['window_count']
        })

    return output

def _append_segment(results, type_code, start_idx, end_idx, scores, target_uid, stride):
    if type_code == 0:
        user = "unknown"
        type_str = "silence"
    elif type_code == 1:
        user = target_uid
        type_str = "target"
    else:
        user = "customer"
        type_str = "other"
    
    results.append({
        "start": start_idx * stride,
        "end": end_idx * stride,
        "type": type_str,
        "user": user,
        "scores": scores,
        "window_count": end_idx - start_idx
    })
