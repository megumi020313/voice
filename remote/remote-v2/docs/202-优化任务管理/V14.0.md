    # ... (保留前面的代码) ...

    def extract_embedding(self, audio, is_batch=False):
        """
        提取特征向量
        :param audio:

            - Batch音频: numpy array (B, T) 或 list of numpy arrays
        :param is_batch: 是否为批量输入

        with torch.no_grad():
            if is_batch:

                embedding = self.model(input_tensor) 
                
            else:

                if isinstance(audio, np.ndarray):
                    input_tensor = torch.from_numpy(audio).float().to(self.device)
                else:
                    input_tensor = audio.to(self.device)

### 第二步：重写 Pipeline (`src/pipeline/inference_pipeline.py`)

这是本次修改的**重中之重**。我们将实现：
1.  **VAD 过滤**：只提取人声部分的窗口。
2.  **Batch 推理**：一次性算出所有窗口的分数。
3.  **细粒度拼接**：基于窗口结果拼接，而不是 VAD 段结果。

```python
import numpy as np
import torch
from src.utils.audio_utils import calculate_diarization_intervals

class InferencePipeline:
    # ... (__init__ 等保持不变) ...

    def run(self, audio_numpy):
        # 1. VAD 获取人声时间戳
        # 建议 min_silence_duration_ms=300 (适中，不要太长也不要太短)
        vad_segments = self.vad_model.get_speech_timestamps(
            audio_numpy, 
            threshold=0.3,
            min_silence_duration_ms=300
        )
        
        if not vad_segments:
            return {"best_user": "unknown", "score": 0.0, "segments": [], "raw_segments": []}

        # 2. 收集所有有效窗口 (Batch Preparation)
        window_size_samples = int(self.config.window_size * 16000)
        stride_samples = int(self.config.window_stride * 16000)
        
        batch_windows = []
        batch_meta = [] # 记录每个窗口的时间信息
        
        for vad in vad_segments:
            start_sample = vad['start']
            end_sample = vad['end']
            
            # 在 VAD 片段内滑动
            for ws in range(start_sample, end_sample - window_size_samples + 1, stride_samples):
                we = ws + window_size_samples
                window = audio_numpy[ws:we]
                
                batch_windows.append(window)
                batch_meta.append({
                    'start': ws / 16000.0,
                    'end': we / 16000.0
                })
        
        if not batch_windows:
             return {"best_user": "unknown", "score": 0.0, "segments": [], "raw_segments": []}

        # 3. Batch 推理 (GPU 加速核心)
        # 将 list 转为 numpy array (B, T)
        batch_input = np.array(batch_windows)
        
        # 一次性提取所有 Embedding
        # embeddings shape: (Batch_Size, 192)
        embeddings = self.model.extract_embedding(batch_input, is_batch=True)
        
        # 4. Batch 评分 (AS-Norm / Cosine)
        # 获取所有注册用户向量
        all_users = self.vector_storage.load_all_vectors_dict()
        
        raw_segments = []
        
        # 遍历每个窗口的结果
        for i, emb in enumerate(embeddings):
            best_win_user = "unknown"
            best_win_score = -999.0
            
            # 与库中所有人比对
            for uid, user_emb in all_users.items():
                # 这里也可以优化为矩阵乘法，但通常用户数不多，循环即可
                score = self.vector_storage.compute_as_norm_score(user_emb, emb)
                
                if score > best_win_score:
                    best_win_score = score
                    best_win_user = uid
            
            # 记录结果
            meta = batch_meta[i]
            raw_segments.append({
                "start": meta['start'],
                "end": meta['end'],
                "user": best_win_user,
                "score": best_win_score
            })

        # 5. 全局统计 (Best User)
        # 统计谁出现的窗口最多，或者平均分最高
        user_stats = {}
        for seg in raw_segments:
            u = seg['user']
            s = seg['score']
            # 只有超过一定阈值才统计，防止噪音干扰
            if s > self.config.low_threshold: 
                if u not in user_stats: user_stats[u] = []
                user_stats[u].append(s)
        
        if user_stats:
            best_user = max(user_stats.items(), key=lambda x: sum(x[1])/len(x[1]))[0]
            best_score = sum(user_stats[best_user])/len(user_stats[best_user])
        else:
            best_user = "unknown"
            best_score = 0.0

        # 6. 生成最终分段 (调用 V12.0 的算法)
        # 注意：这里传入的是 raw_segments (窗口级)，不是 VAD 级
        total_duration = len(audio_numpy) / 16000.0
        
        # 提高识别阈值，解决误识问题
        ident_threshold = getattr(self.config, 'low_threshold', 2.5) # 建议提高到 2.5
        
        final_segments = calculate_diarization_intervals(
            raw_segments=raw_segments,
            total_duration=total_duration,
            stride=self.config.window_stride,
            max_silence_gap=15.0,
            identification_threshold=ident_threshold
        )
        
        return {
            "best_user": best_user,
            "score": best_score,
            "status": "success",
            "segments": final_segments,
            "raw_segments": raw_segments
        }
```

    *   **效果**：VAD 层面保证了“一句话”是连贯的，不会把中间的停顿切开。只有真正的长静音才会让 VAD 断开。

#### 原因 B：分数抖动（中间某 0.1s 分数掉下去了）
*   **旧情况**：直接看窗口分数，低了就变 Unknown/Silence。
*   **V14.0 解法**：依赖 `calculate_diarization_intervals` 里的 **“网格平滑 (Smoothing)”** 逻辑。
    *   **代码回顾**：V12.0/V9.5 的算法里有一个 `fill_gap_steps` 循环。
    *   **效果**：即使中间有 3-5 个窗口（0.3s-0.5s）分数掉到了阈值以下，平滑逻辑会强制把它们“涂”回 Target。

| 特性 | V13 (VAD 投票法) | V14 (窗口 Batch 法) |
